cmake_minimum_required(VERSION 3.24)
project(photo_denoise VERSION 1.2.0 LANGUAGES C)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This version requires Windows and the MSVC x64 toolchain.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(ORT_ROOT "${CMAKE_SOURCE_DIR}/third_party/onnxruntime" CACHE PATH "Extracted ONNX Runtime NuGet")
set(DML_ROOT "${CMAKE_SOURCE_DIR}/third_party/directml" CACHE PATH "Extracted DirectML NuGet")
set(LIBRAW_ROOT "${CMAKE_SOURCE_DIR}/third_party/libraw/LibRaw-0.22.3" CACHE PATH "LibRaw Windows SDK")
if(NOT EXISTS "${ORT_ROOT}/build/native/include/onnxruntime_c_api.h")
  message(FATAL_ERROR "Run scripts/setup.ps1 first")
endif()
add_library(denoise_core STATIC src/common.c src/image/storage.c src/image/load.c
  src/image/raw.c src/image/wic.c src/image/tiff_writer.c src/model/catalog.c
  src/model/download.c src/backend/ort.c src/pipeline/denoise.c)
target_include_directories(denoise_core PUBLIC include PRIVATE "${ORT_ROOT}/build/native/include" "${LIBRAW_ROOT}")
target_compile_definitions(denoise_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX COBJMACROS _CRT_SECURE_NO_WARNINGS)
target_compile_options(denoise_core PRIVATE /W4 /utf-8)
target_link_libraries(denoise_core PUBLIC "${ORT_ROOT}/runtimes/win-x64/native/onnxruntime.lib"
  "${LIBRAW_ROOT}/lib/libraw.lib" ole32 uuid dxguid dxgi d3d12 winhttp bcrypt windowscodecs mscms propsys)
add_library(denoise_app STATIC src/app/job.c src/tui/console.c)
target_link_libraries(denoise_app PUBLIC denoise_core)
target_compile_options(denoise_app PRIVATE /W4 /utf-8)
add_executable(denoise src/cli/main.c)
target_link_libraries(denoise PRIVATE denoise_app)
target_compile_options(denoise PRIVATE /W4 /utf-8)
set_target_properties(denoise PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(denoise PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_custom_command(TARGET denoise POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT_ROOT}/runtimes/win-x64/native/onnxruntime.dll" "$<TARGET_FILE_DIR:denoise>"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DML_ROOT}/bin/x64-win/DirectML.dll" "$<TARGET_FILE_DIR:denoise>")
add_custom_command(TARGET denoise POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${LIBRAW_ROOT}/bin/libraw.dll" "$<TARGET_FILE_DIR:denoise>")
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.c)
  target_link_libraries(core_tests PRIVATE denoise_core)
  set_target_properties(core_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(core_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  endforeach()
  add_test(NAME core COMMAND core_tests)
  add_executable(io_roundtrip tests/io_roundtrip.c)
  target_link_libraries(io_roundtrip PRIVATE denoise_core)
  set_target_properties(io_roundtrip PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(io_roundtrip PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  endforeach()
  add_test(NAME cli_help COMMAND denoise --help)
  add_test(NAME cli_model_list COMMAND denoise model list)
  add_test(NAME cli_invalid_amount COMMAND denoise missing.png 101)
  set_tests_properties(cli_invalid_amount PROPERTIES WILL_FAIL TRUE)
endif()
